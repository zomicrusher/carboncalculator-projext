/** * Dynamic Carbon Footprint Calculator Logic * Handles step navigation, input reading, calculation rules, and dynamic report updating. */ // Array of Tab IDs representing sequential steps const tabList = [ 'general-tab', 'home-tab', 'travel-tab', 'food-tab', 'electronic-tab', 'shopping-tab', 'result-tab' ]; let currentTabIndex = 0; // Country baseline factors in Tonnes CO2e/year const countryFactors = { 'Saudi Arabia': 1.8, 'United Arab Emirates': 1.9, 'Bahrain': 1.6, 'Kuwait': 1.7, 'Oman': 1.4, 'Qatar': 2.0, 'Other': 1.2 }; // Diet emissions in Tonnes CO2e/year const dietFactors = { 'meat-heavy': 2.8, 'average': 2.0, 'vegetarian': 1.4, 'vegan': 1.0 }; // Shopping multiplier in Tonnes CO2e/year const shoppingFactors = { 'minimal': 0.4, 'moderate': 0.9, 'frequent': 1.6 }; /** * Switch active step tab */ function switchTab(tabId) { const tabElement = document.getElementById(tabId); if (tabElement) { const bsTab = new bootstrap.Tab(tabElement); bsTab.show(); currentTabIndex = tabList.indexOf(tabId); updateButtonStates(); calculateCarbonFootprint(); } } /** * Navigate through steps (direction: +1 for Next, -1 for Prev) */ function navigateStep(direction) { let newIndex = currentTabIndex + direction; if (newIndex >= 0 && newIndex < tabList.length) { switchTab(tabList[newIndex]); } } /** * Update Progress Bar, Buttons, and Header text based on current tab index */ function updateButtonStates() { const prevBtn = document.getElementById('prevBtn'); const nextBtn = document.getElementById('nextBtn'); const progressBar = document.getElementById('progressBar'); const cardHeaderTitle = document.getElementById('cardHeaderTitle'); // Update progress bar percentage const progressPercent = Math.round(((currentTabIndex + 1) / tabList.length) * 100); if (progressBar) { progressBar.style.width = `${progressPercent}%`; } // Update Header step text if (cardHeaderTitle) { if (currentTabIndex === tabList.length - 1) { cardHeaderTitle.innerText = "Your Personal Carbon Assessment Result"; } else { cardHeaderTitle.innerText = `Fill the form below - Step ${currentTabIndex + 1} of ${tabList.length}`; } } // Disable Prev button if on the first step if (prevBtn) { prevBtn.disabled = (currentTabIndex === 0); } // Change Next button text on final steps if (nextBtn) { if (currentTabIndex === tabList.length - 2) { nextBtn.innerHTML = 'Calculate '; } else if (currentTabIndex === tabList.length - 1) { nextBtn.innerHTML = 'Finish '; } else { nextBtn.innerHTML = 'Next '; } } } /** * Main Calculation Logic * Evaluates inputs across all forms and updates result fields */ function calculateCarbonFootprint() { // 1. General Profile Inputs const nameInput = document.getElementById('userName')?.value.trim(); const userName = nameInput || "Eco-Warrior"; const userAge = parseInt(document.getElementById('userAge')?.value) || 25; const userCountry = document.getElementById('userCountry')?.value || 'Saudi Arabia'; const countryBase = countryFactors[userCountry] || 1.2; // 2. Household Energy Calculation const electricityKWh = parseFloat(document.getElementById('electricity')?.value) || 0; const householdMembers = Math.max(1, parseInt(document.getElementById('householdMembers')?.value) || 1); // Formula: (kWh per month * 12 months * 0.0005 Tonnes CO2/kWh) / shared household members const homeEmissions = ((electricityKWh * 12 * 0.0005) / householdMembers); // 3. Travel & Transportation const carKmWeek = parseFloat(document.getElementById('carKm')?.value) || 0; const flightsPerYear = parseFloat(document.getElementById('flights')?.value) || 0; // Formula: (km/wk * 52 wk * 0.00017 Tonnes CO2/km) + (flights * 0.5 Tonnes per flight) const travelEmissions = (carKmWeek * 52 * 0.00017) + (flightsPerYear * 0.5); // 4. Food Choices const dietType = document.getElementById('dietType')?.value || 'average'; const foodEmissions = dietFactors[dietType] || 2.0; // 5. Electronics & Digital Usage const screenHours = parseFloat(document.getElementById('screenHours')?.value) || 0; // Formula: screen hours * 365 days * 0.0001 Tonnes CO2/hr const electronicsEmissions = screenHours * 365 * 0.0001; // 6. Shopping & Consumption const shoppingHabit = document.getElementById('shoppingHabit')?.value || 'moderate'; const shoppingEmissions = shoppingFactors[shoppingHabit] || 0.9; // Total Footprint Calculation const totalEmissions = homeEmissions + travelEmissions + foodEmissions + electronicsEmissions + shoppingEmissions + countryBase; // --- Update UI Display --- const co2Output = document.getElementById('co2-output'); if (co2Output) { co2Output.innerText = totalEmissions.toFixed(2); } // Update Greeting & Summary const greetingElem = document.getElementById('user-greeting'); if (greetingElem) { greetingElem.innerText = `Hello ${userName}!`; } const summaryElem = document.getElementById('summary-statement'); if (summaryElem) { summaryElem.innerText = `Based on your profile as a ${userAge}-year-old residing in ${userCountry}, your total annual carbon footprint is estimated at ${totalEmissions.toFixed(2)} Tonnes CO2e/year.`; } // Update Breakdown Display Cards const bkHome = document.getElementById('bk-home'); const bkTravel = document.getElementById('bk-travel'); const bkFood = document.getElementById('bk-food'); const bkElec = document.getElementById('bk-electronics'); const bkShop = document.getElementById('bk-shopping'); const bkCty = document.getElementById('bk-country'); if (bkHome) bkHome.innerText = `${homeEmissions.toFixed(2)} t`; if (bkTravel) bkTravel.innerText = `${travelEmissions.toFixed(2)} t`; if (bkFood) bkFood.innerText = `${foodEmissions.toFixed(2)} t`; if (bkElec) bkElec.innerText = `${electronicsEmissions.toFixed(2)} t`; if (bkShop) bkShop.innerText = `${shoppingEmissions.toFixed(2)} t`; if (bkCty) bkCty.innerText = `${countryBase.toFixed(2)} t`; } // Attach input listeners across forms for real-time automatic calculation updates document.addEventListener('DOMContentLoaded', () => { const inputs = document.querySelectorAll('input, select'); inputs.forEach(input => { input.addEventListener('input', calculateCarbonFootprint); input.addEventListener('change', calculateCarbonFootprint); }); // Synchronize tab clicks document.querySelectorAll('button[data-bs-toggle="tab"]').forEach((tabBtn, index) => { tabBtn.addEventListener('shown.bs.tab', () => { currentTabIndex = index; updateButtonStates(); calculateCarbonFootprint(); }); }); // Initial setup on load updateButtonStates(); calculateCarbonFootprint(); });
